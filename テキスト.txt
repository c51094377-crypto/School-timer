<script>

const remaining = document.getElementById("remaining");
const statusText = document.getElementById("status");
const progressBar = document.getElementById("progressBar");
const percentText = document.getElementById("percent");

const lesson50 = document.getElementById("lesson50");
const lesson45 = document.getElementById("lesson45");
const startButton = document.getElementById("start");
const resetButton = document.getElementById("reset");

let selectedMinutes = null;
let timer = null;


/* =========================
   授業時間選択
========================= */

lesson50.addEventListener("click", () => {

  selectedMinutes = 50;

  lesson50.classList.add("active");
  lesson45.classList.remove("active");

});


lesson45.addEventListener("click", () => {

  selectedMinutes = 45;

  lesson45.classList.add("active");
  lesson50.classList.remove("active");

});


/* =========================
   時刻計算
========================= */

function updateTimer() {

  const startTime =
    Number(localStorage.getItem("classStartTime"));

  const duration =
    Number(localStorage.getItem("classDuration"));

  if (!startTime || !duration) {

    remaining.textContent = "--:--";
    statusText.textContent =
      "授業時間を選択してください";

    return;

  }


  const now = Date.now();

  const durationMs =
    duration * 60 * 1000;

  const endTime =
    startTime + durationMs;

  const remainingMs =
    endTime - now;


  /* =========================
     授業終了
  ========================= */

  if (remainingMs <= 0) {

    remaining.textContent = "00:00";

    statusText.textContent =
      "🎉 授業終了！";

    progressBar.style.width = "100%";

    percentText.textContent =
      "進捗 100%";

    clearInterval(timer);

    return;

  }


  /* =========================
     残り時間
  ========================= */

  const totalSeconds =
    Math.ceil(remainingMs / 1000);

  const minutes =
    Math.floor(totalSeconds / 60);

  const seconds =
    totalSeconds % 60;


  remaining.textContent =
    String(minutes).padStart(2, "0")
    + ":"
    + String(seconds).padStart(2, "0");


  /* =========================
     進捗率
  ========================= */

  const elapsed =
    now - startTime;

  const progress =
    Math.min(
      100,
      Math.max(
        0,
        elapsed / durationMs * 100
      )
    );


  progressBar.style.width =
    progress + "%";

  percentText.textContent =
    "進捗 " + progress.toFixed(1) + "%";


  statusText.textContent =
    duration + "分授業";
}


/* =========================
   授業開始
========================= */

startButton.addEventListener("click", () => {

  if (!selectedMinutes) {

    alert("45分または50分を選択してください。");

    return;

  }


  const startTime = Date.now();


  /*
    現在時刻と授業時間を保存

    Date.now()は1970年からの経過ミリ秒なので、
    ページを閉じても時間そのものは失われない。
  */

  localStorage.setItem(
    "classStartTime",
    String(startTime)
  );

  localStorage.setItem(
    "classDuration",
    String(selectedMinutes)
  );


  clearInterval(timer);

  updateTimer();

  timer = setInterval(
    updateTimer,
    200
  );

});


/* =========================
   リセット
========================= */

resetButton.addEventListener("click", () => {

  localStorage.removeItem("classStartTime");
  localStorage.removeItem("classDuration");

  clearInterval(timer);

  remaining.textContent = "--:--";

  statusText.textContent =
    "授業時間を選択してください";

  progressBar.style.width = "0%";

  percentText.textContent =
    "進捗 0%";

});


/* =========================
   ページを開いた時
========================= */

function restoreTimer() {

  const savedStart =
    localStorage.getItem("classStartTime");

  const savedDuration =
    localStorage.getItem("classDuration");


  if (!savedStart || !savedDuration) {

    return;

  }


  selectedMinutes =
    Number(savedDuration);


  if (selectedMinutes === 50) {

    lesson50.classList.add("active");

  }

  if (selectedMinutes === 45) {

    lesson45.classList.add("active");

  }


  /*
    保存された開始時刻と
    現在時刻を比較して再計算
  */

  updateTimer();


  timer = setInterval(
    updateTimer,
    200
  );

}


/* 起動 */

restoreTimer();

</script>
